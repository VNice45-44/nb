public.machine_components

    id uuid PK default extensions.uuid_generate_v4()
    machine_id FK → machines(id) on delete cascade
    owner_id FK → profiles(id) on delete cascade
    parent_id self-FK nullable on delete set null
    title, component_kind required
    properties jsonb default '{}'
    status text default 'concept'
    sort_order integer default 0
    created_at, updated_at

Indexes on:

    machine_id, parent_id, component_kind

public.machine_component_links

    id uuid PK
    machine_id FK → machines(id) on delete cascade
    source_component_id + target_component_id FKs → machine_components(id) on delete cascade
    relation_kind text default 'connects'
    metadata jsonb default '{}'
    created_at
    Added constraint: no self-links (source_component_id <> target_component_id)

Indexes on:

    machine_id, source_component_id, target_component_id
